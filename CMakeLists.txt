cmake_minimum_required(VERSION 3.20)
project(SonicSync LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()

include(shared/presets/catalog.cmake)
add_library(sonicsync_dsp shared/dsp/dsp.cpp shared/dsp/eq_state.cpp)
target_include_directories(sonicsync_dsp PUBLIC shared/dsp "${CMAKE_CURRENT_BINARY_DIR}/generated")
function(sonicsync_enable_warnings target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive- /EHsc /utf-8)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
  endif()
endfunction()
sonicsync_enable_warnings(sonicsync_dsp)

add_executable(dsp_benchmark tools/dsp_benchmark.cpp)
target_link_libraries(dsp_benchmark PRIVATE sonicsync_dsp)
sonicsync_enable_warnings(dsp_benchmark)

include(CTest)
if(BUILD_TESTING)
  add_executable(eq_acceptance tests/eq_acceptance.cpp)
  target_link_libraries(eq_acceptance PRIVATE sonicsync_dsp)
  add_test(NAME eq_acceptance COMMAND eq_acceptance)
  add_executable(dsp_tests tests/dsp_tests.cpp)
  target_link_libraries(dsp_tests PRIVATE sonicsync_dsp)
  sonicsync_enable_warnings(dsp_tests)
  add_test(NAME dsp_tests COMMAND dsp_tests)
  add_executable(windows_library_tests tests/windows_library_tests.cpp)
  sonicsync_enable_warnings(windows_library_tests)
  add_test(NAME windows_library_tests COMMAND windows_library_tests)
endif()

if(WIN32)
  enable_language(RC)
  add_custom_target(sonicsync_virtual_endpoint
    COMMAND powershell.exe -NoProfile -ExecutionPolicy Bypass
      -File "${CMAKE_CURRENT_SOURCE_DIR}/tools/build_sonicsync_virtual_endpoint.ps1"
      -OutputDirectory "${CMAKE_CURRENT_BINARY_DIR}/driver"
    COMMENT "Building the SonicSync SysVAD-derived virtual render endpoint (no install/signing)"
    VERBATIM
  )
  add_executable(sonicsync_wasapi platforms/windows/wasapi_bridge.cpp)
  target_link_libraries(sonicsync_wasapi PRIVATE sonicsync_dsp ole32 uuid avrt)
  sonicsync_enable_warnings(sonicsync_wasapi)
  add_executable(sonicsync_virtual_endpoint_relay platforms/windows/wasapi_bridge.cpp)
  target_compile_definitions(sonicsync_virtual_endpoint_relay PRIVATE SONICSYNC_VIRTUAL_ENDPOINT_RELAY=1)
  target_link_libraries(sonicsync_virtual_endpoint_relay PRIVATE sonicsync_dsp ole32 uuid avrt)
  sonicsync_enable_warnings(sonicsync_virtual_endpoint_relay)
  add_executable(sonicsync_player platforms/windows/wasapi_player.cpp)
  target_link_libraries(sonicsync_player PRIVATE sonicsync_dsp ole32 uuid avrt)
  sonicsync_enable_warnings(sonicsync_player)
  file(TO_CMAKE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/assets/brand/sonicsync.ico" SONICSYNC_ICON_FILE)
  configure_file(platforms/windows/sonicsync_gui.rc.in "${CMAKE_CURRENT_BINARY_DIR}/sonicsync_gui.rc" @ONLY)
  add_executable(sonicsync_gui WIN32 platforms/windows/gui.cpp platforms/windows/artwork.cpp platforms/windows/studio_session.cpp platforms/windows/resource.h "${CMAKE_CURRENT_BINARY_DIR}/sonicsync_gui.rc")
  target_include_directories(sonicsync_gui PRIVATE shared/dsp platforms/windows)
  find_path(SONICSYNC_CPPWINRT_INCLUDE winrt/Windows.Media.Control.h
    HINTS "C:/Program Files (x86)/Windows Kits/10/Include/${CMAKE_VS_WINDOWS_TARGET_PLATFORM_VERSION}/cppwinrt"
    ENV INCLUDE)
  if(EXISTS "${SONICSYNC_CPPWINRT_INCLUDE}/winrt/Windows.Media.Control.h")
    target_include_directories(sonicsync_gui PRIVATE "${SONICSYNC_CPPWINRT_INCLUDE}")
    add_executable(sonicsync_media_session_remote platforms/windows/media_session_remote.cpp)
    target_include_directories(sonicsync_media_session_remote PRIVATE "${SONICSYNC_CPPWINRT_INCLUDE}")
    target_link_libraries(sonicsync_media_session_remote PRIVATE windowsapp)
    sonicsync_enable_warnings(sonicsync_media_session_remote)
    if(BUILD_TESTING)
      add_test(NAME windows_media_session_remote_test COMMAND sonicsync_media_session_remote --state)
    endif()
  else()
    message(FATAL_ERROR "Windows SDK C++/WinRT headers are required for the media-session UI: ${SONICSYNC_CPPWINRT_INCLUDE}")
  endif()
  target_link_libraries(sonicsync_gui PRIVATE sonicsync_dsp comctl32 comdlg32 ole32 uuid shell32 windowsapp gdiplus windowscodecs)
  sonicsync_enable_warnings(sonicsync_gui)

  if(BUILD_TESTING)
    add_executable(windows_wave_tests tests/windows_wave_tests.cpp)
    target_link_libraries(windows_wave_tests PRIVATE uuid)
    sonicsync_enable_warnings(windows_wave_tests)
    add_test(NAME windows_wave_tests COMMAND windows_wave_tests)
    add_executable(windows_session_tests tests/windows_session_tests.cpp platforms/windows/studio_session.cpp)
    target_include_directories(windows_session_tests PRIVATE platforms/windows)
    sonicsync_enable_warnings(windows_session_tests)
    add_test(NAME windows_session_tests COMMAND windows_session_tests)
    add_executable(windows_artwork_tests tests/windows_artwork_tests.cpp platforms/windows/artwork.cpp)
    target_include_directories(windows_artwork_tests PRIVATE platforms/windows)
    target_link_libraries(windows_artwork_tests PRIVATE windowscodecs ole32 uuid)
    target_compile_definitions(windows_artwork_tests PRIVATE SONICSYNC_TEST_IMAGE_PATH="${CMAKE_CURRENT_SOURCE_DIR}/assets/brand/sonicsync-mark-1024.png")
    sonicsync_enable_warnings(windows_artwork_tests)
    add_test(NAME windows_artwork_tests COMMAND windows_artwork_tests)
  endif()

  install(TARGETS sonicsync_gui sonicsync_player sonicsync_wasapi sonicsync_virtual_endpoint_relay sonicsync_media_session_remote RUNTIME DESTINATION .)
